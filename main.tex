\section{The Vertical Structure Operator}
\label{sec:building_L}
We begin with the 3D primitive equations for a statically stable atmosphere in log-pressure coordinates $\z$
%
\footnote{In log-pressure coordinates, the vertical coordinate is expressed as $\z=H_s\ln(p/p_s)$, where $p_s$ is the pressure that defines the surface ($\z=0$) and $H_s=RT_0/g$ is a reference scale height of the atmosphere}:
%
\begin{subequations}
\label{eq:linear_system}
\begin{align}
\frac{\partial u}{\partial t} - f v
    &= -\frac{\partial \Phi}{\partial x},
    \label{eq:u_momentum} \\
\frac{\partial v}{\partial t} + f u
    &= -\frac{\partial \Phi}{\partial y},
    \label{eq:v_momentum} \\
\frac{\partial u}{\partial x}
+ \frac{\partial v}{\partial y}
+ \frac{1}{\rho_0}
  \frac{\partial}{\partial \z}
  \left( \rho_0 \w \right)
    &= 0,
    \label{eq:continuity} \\
\frac{\partial \Phi}{\partial \z}
    &= \frac{R T}{H_s},
    \label{eq:hydrostatic} \\
\frac{\partial}{\partial t}
\left(
    \frac{\partial \Phi}{\partial \z}
\right)
+ N^2 \w
    &= 0,
    \label{eq:thermodynamic}
\end{align}
\end{subequations}
%
where $\z$ is our log-pressure vertical coordinate; $\w$ is the vertical velocity in our log-pressure coordinates; $\rho_0(\z)$ is a fixed reference density profile; $T$ is the temperature perturbation from a background profile $T_0(\z)$; $N^2(\z)>0$ is the buoyancy frequency, a measure of the background stratification; and $\Phi$ is the geopotential. The other variables take their standard meteorological meanings.
%
\par We also impose rigid lid boundary conditions at our a height $\z=\z_B$ and a top height $\z=\z_T$
%
\begin{subequations}
\begin{align}
  \label{eq:rigid_lid}
\w(x,y,t;\z_B) &= 0, \\
\w(x,y,t;\z_T) &= 0,
\end{align}
\end{subequations}
%
%
\begin{exercise}
The first two equations of \eqref{eq:linear_system} are the zonal and meridional momentum equations. What are the corresponding names for the other three?
\end{exercise}
%
Our goal is to better understand the vertical behavior of this model. Therefore, it would be useful to collapse all the vertical structure information into a single term. The vertical coordinate appears in \eqref{eq:continuity}-\eqref{eq:thermodynamic}. These equations involve $u$, $v$, $w$, $T$, and $\phi$. The horizontal equations are just in terms of $u$, $v$, and $\phi$, so we should try to eliminate $w$ and $T$ by writing them in terms of the other three variables. 
%
\par From the thermodynamic equation \eqref{eq:hydrostatic} we have
\begin{equation}
    \w = \frac{-1}{N^2}\frac{\partial}{\partial t}\left(\frac{\partial \Phi}{\partial \z}\right).
\end{equation}
%
Multiplying by $\rho_0$ and taking the vertical derivative gives
\begin{equation}
    \frac{\partial}{\partial \z}\left(\rho_0 \w\right) = -\frac{\partial}{\partial\z}\left(\frac{\rho_0}{N^2}\frac{\partial}{\partial t}\left(\frac{\partial\Phi}{\partial \z}\right)\right).
\end{equation}
%
By the continuity equation \eqref{eq:continuity}, this is also equal to 
\begin{equation}
\frac{\partial}{\partial \z}(\rho_0\w) = -\rho_0\left(\frac{\partial u}{\partial x} + \frac{\partial v}{\partial y}\right).
\end{equation}
%
Equating these two expressions gives
\begin{equation}
  -\rho_0\left(\frac{\partial u}{\partial x} + \frac{\partial v}{\partial y}\right) = -\frac{\partial}{\partial\z}\left(\frac{\rho_0}{N^2}\frac{\partial}{\partial t}\left(\frac{\partial\Phi}{\partial \z}\right)\right).
\end{equation}
%
Rearranging, 
\begin{equation}
  \frac{1}{\rho_0}\frac{\partial}{\partial\z}\left(\frac{\rho_0}{N^2}\frac{\partial}{\partial t}\frac{\partial\Phi}{\partial\z}\right) - \left(\frac{\partial u}{\partial x} + \frac{\partial v}{\partial y}\right) = 0.
\end{equation}
%
Assuming that $N^2$ is constant in time, we can pull the time derivative out to the front of the vertical derivatives and re-arrange to get
\begin{equation}
   \frac{\partial}{\partial t}\left[\frac{1}{\rho_0}\frac{\partial}{\partial\z}\left(\frac{\rho_0}{N^2}\frac{\partial\Phi}{\partial\z}\right)\right]- \left(\frac{\partial u}{\partial x} + \frac{\partial v}{\partial y}\right) = 0.
\end{equation}
%
Note that the bracketed term contains all the vertical derivatives now. We can thus define an operator
%
\footnote{An operator is a map that takes a function as and input and returns another function as an output. A common example is the derivative operator $d/dx$, which maps a function $f(x)$ to its derivative $f'(x)$.}
%
$L$, which we term the \textit{vertical structure operator}, which acts on a function $\Phi(\z)$ as
\begin{equation}
  \boxed{
  \label{eq:L_def}
  L[\Phi] = \frac{1}{\rho_0}\frac{\partial}{\partial\z}\left(\frac{\rho_0}{N^2}\frac{\partial\Phi}{\partial\z}\right).
  }
\end{equation}
%
Thus, our system of equations can now be written as
\begin{subequations}
\label{eq:system_with_L}
\begin{align}
\frac{\partial u}{\partial t} - f v &= -\frac{\partial \Phi}{\partial x},
    \label{eq:linear_system_reduced_a} \\
\frac{\partial v}{\partial t} + f u &= -\frac{\partial \Phi}{\partial y},
    \label{eq:linear_system_reduced_b} \\
\frac{\partial}{\partial t}L[\Phi] &= \left(\frac{\partial u}{\partial x} + \frac{\partial v}{\partial y}\right) 
    \label{eq:linear_system_reduced_c}
\end{align}
\end{subequations}
%

%
\section{Vertical Normal Modes as an Eigenbasis}
\subsection{What good would an eigenfunction do?}
In writing \eqref{eq:system_with_L}, our zonal and meridional momentum equations are the same as before, but we have now combined all the vertical derivatives into a single term, $L[\Phi]$. To understand how the vertical behavior of this system, then, we need to understand how $L$ acts on functions. 
%
\begin{exercise}
   \label{exercise:L_linear}
  Show that $L$ is a linear operator. That is, for any functions $f(\z)$ and $g(\z)$, and constants $\alpha$ and $\beta$ show that
\begin{equation}
  L[\alpha f(\z)+\beta g(\z)]=\alpha L[f] + \beta L[g]
\end{equation}
\end{exercise}
%
\par A good starting point to solving any problem is to think about what conditions would make the problem easier to solve. In that spirit, let's suppose (for now, just suppose) that there were, somewhere in the world, some function $\psi(\z)$ such that 
%
\begin{equation}
L[\psi] = -\lambda \psi,
\label{eq:eigenfunction_definition}
\end{equation}
%
for some constant $\lambda\in\mathbb{R}$ (the reason for the negative sign in Equation \ref{eq:eigenfunction_definition} will become apparent soon). Such a function $\psi$ would be called an ``eigenfunction" of $L$, and $\lambda$ its corresponding ``eigenvalue"
%
\footnote{The root word ``eigen'' mean ``own", or ``self" in German. The notion of an eigenfunction is analogous to notion of an eigenvector $\vec v$ of a matrix $A$, which satisfies $A\vec v = c\vec v$ for some constant $c$. }
%
\begin{exercise}
  \label{exercise:scaled_eigfn}
  Show that if $\psi(\z)$ is an eigenfunction of $L$ with corresponding eigenvalue $\lambda$, then for any constant $c$, the scaled function
  $$\varphi(p)=c\psi(\z)$$
  is also an eigenvector with eigenvalue $\lambda$.
\end{exercise}
%
While we are supposing things, let's also suppose that this function $\psi$ were to also give us the vertical structures of $u$, $v$, and $\Phi$. That is, suppose (again, just supposing for now) that
%
\begin{subequations}
  \label{eq:eigenstructures}
\begin{align}
  u(x,y,\z,t) = \hat u(x,y,t)\psi(\z)\\
  v(x,y,\z,t) = \hat v(x,y,t)\psi(\z), \\
  \Phi(x,y,\z,t) = \hat \Phi(x,y,t)\psi(\z).
\end{align}
\end{subequations}
%
\shallowwatermarker
If all of this were so, then we could write our system of equations \eqref{eq:system_with_L} as
\begin{subequations}
  \label{eq:swe_system}
\begin{align}
\frac{\partial \hat u}{\partial t} - f \hat v &= -\frac{\partial \hat \Phi}{\partial x}, \label{eq:swe_u} \\
\frac{\partial \hat v}{\partial t} + f \hat u &= -\frac{\partial \hat \Phi}{\partial y}, \label{eq:swe_v} \\
\frac{\partial\hat\Phi}{\partial t} + \frac{1}{\lambda}\left(\frac{\partial \hat u}{\partial x} + \frac{\partial \hat v}{\partial y}\right) &= 0. \label{eq:swe_phi}
\end{align}
\end{subequations}
%
%
\begin{exercise}
Using \eqref{eq:eigenfunction_definition}, and assuming $u$, $v$, and $\Phi$ have the form given by \eqref{eq:eigenstructures}, derive Equations \eqref{eq:swe_u}-\eqref{eq:swe_phi}.
\end{exercise}
%
We observe that the system \eqref{eq:swe_system} is of the same form as the shallow water equations for a layer of incompressible fluid. In this pseudo-shallow water system, the role of the gravity wave speed $c$ is played by the $\lambda^{-1/2}$. Since in a shallow water system the gravity wave speed is given by $c=\sqrt{gh}$, where $h$ is the depth of the fluid, we can define a parameter called the \textit{equivalent depth} as 
\begin{equation}
  \label{eq:equivalent_depth}
  \boxed{
  h_e=\frac{1}{g\lambda}
  }
\end{equation}
%
Let us summarize what we have shown up till now. We have shown that 
\begin{itemize}
\item if $L[\Phi]$ had an eigenfunction $\psi(\z)$ with corresponding eigenvaue $\lambda$; \textbf{and}
\item if the vertical structures of $u$, $v$, and $\Phi$ were all given by that eigenfunction $\psi$;
\item\textbf{then} our original three-dimensional system would behave like a two-dimensional incompressible fluid, whose gravity wave speed is given by $\lambda^{-1}$. 
\end{itemize}
%
\par The important point here is that, when the above conditions are satisfied, the whole systems behaves \textit{as though it had no vertical dimension at all}. This is a massive simplification of the dynamics. So given that the existence of these eigenfunctions would make our lives much easier, we are motivated to ask 
\begin{enumerate}
  \item Do such eigenfunctions exist? How many are there?
  \item When can the vertical structure of $u$ and $v$ be given by eigenfunctions of $\Phi$?
  \item How do the eigenvalues depend on the physical parameters of our problem?
  \item How can we compute these eigenfunctions?
\end{enumerate}
In the remainder of this section, we will answer Questions 1 and 2. We will answer Question 3 by explicitly computing the eigenfunctions in the case that $N^2$ is a constant. Then we will answer Question 4 by discretizing our problem into a generalized eigenvalue problem. 
%
\subsection{About these eigenfunctions}
We now consider the question: how many eigenfunctions does $L$ have, and what properties do they possess? This is a purely mathematical question that depends on $L$ itself. The answer to this question is given by ``Sturm-Liouville'' theory, which is concerned with finding functions $y(x)$ and numbers $\lambda$ that satisfy the differential equation
\begin{equation}
  \label{eq:sturm_liouville}
  \frac{d}{dx}\left[p(x)\frac{dy}{dx}\right] + q(x)y = -\lambda w(x) y
\end{equation}
defined on a bounded interval $[a, b]$, where $p(x)>0$ and $w(x)>0$ for all $x\in[a,b]$, and where the problem satisfies the boundary conditions
\begin{equation}
  \label{eq:robin_conditions}
 \begin{cases}
\alpha_1 y(a) + \alpha_2 y'(a) = 0, 
& \alpha_1,\alpha_2 \text{ not both } 0, \\[4pt]
\beta_1 y(b) + \beta_2 y'(b) = 0, 
& \beta_1,\beta_2 \text{ not both } 0.
\end{cases}
\end{equation}

\begin{exerciseblock}
  \begin{problem}
  Show that \eqref{eq:linear_system_reduced_c} can be written in the form of \eqref{eq:sturm_liouville}. 
  \end{problem}
  \begin{problem}
  \label{ex:boundary_conditions_on_psi}
  \usedlater
  Show that the rigid-lid boundary conditions \eqref{eq:rigid_lid} imply that the eigenfunctions of $L$ satisfy the following boundary conditions:
  \begin{equation}
    \label{eq:boundary_conditions_on_psi}
    \left.\frac{d\psi}{d\z}\right|_{\z_B} = \left.\frac{d\psi}{d\z}\right|_{\z_T} = 0
  \end{equation}
  Verify that these boundary conditions are of the form \eqref{eq:robin_conditions}.
  %
  \\\textit{Hint: Use the thermodynamic equation \eqref{eq:thermodynamic} and separation of variables $\Phi(x,y,z,t)=\phi(x,y,t)\psi(z)$}
  \end{problem}
\end{exerciseblock}
%
Since our problem is of the form \eqref{eq:sturm_liouville}, we can import the following facts about our vertical structure operator $L$, which we will state without proof:
%
\begin{enumerate}
  \item \textbf{$L$ has infinitely many real eigenvalues and eigenfunctions}: There exists an infinite sequence of non-negative real numbers
\begin{equation}
  0\leq\lambda_0<\lambda_1<\dots\lambda_n<\lambda_{n+1}<\dots
\end{equation}
with $\lambda_n\to\infty$ as $n\to\infty$, and a corresponding set of functions $\{\psi_i(\z)\}_{i=0}^\infty$, that are distinct up to scaling by a constant, such that
\begin{equation}
  L[\psi_n(\z)] = -\lambda_n\psi_n(\z)
\end{equation}
  \item \textbf{The eigenfunctions of $L$ are orthogonal under a weighted product}: If we write the mass weighted integral product of two functions $f(\z)$ and $g(\z)$ as 
\begin{equation}
  \label{eq:inner_product_def}
  \langle f, g\rangle = \int_{\z_B}^{\z_T}\rho_0(\z)f(\z)g(\z)\ d\z
\end{equation}
Then the eigenfunctions $\{\psi_i\}$ satisfy
\begin{equation}
  \label{eq:orthogonality}
  \langle\psi_m, \psi_n\rangle =
  \begin{cases}
    0 ,\ m\neq n \\
    1 ,\ m=n
  \end{cases}
\end{equation}
%
A consequence of this is that the eigenfunctions are \textit{linearly
independent}: a weighted sum of them vanishes only if every coefficient
vanishes,
\begin{equation}
\label{eq:lin_indep}
\sum_{i=0}^{N}c_i\psi_i=0
\iff
c_i=0 \ \text{ for every } i.
\end{equation}
%
\item \textbf{The eigenfunctions of $L$ form a complete basis}. We can expand any\footnote{There are pathological functions one can propose that would not satisfy this, but they are not likely to appear in any physically meaningful setting.}
%
function $f(\z)$ in our \textit{eigenbasis} as 
\begin{equation}
  f(\z) = \sum_{i=0}^{\infty}\hat{f_i}\psi_i(\z)
\end{equation}
Where the real numbers $\{\hat{f_i}\}$ are given as 
\begin{equation}
  \hat{f_i} = \langle f(\z), \psi_i(\z)\rangle
\end{equation}
\end{enumerate}
%
These facts about $L$ will allow us to describe out three-dimensional system \eqref{eq:linear_system} as a combination of two-dimensional shallow water systems of the form \eqref{eq:swe_system}.
%
\begin{exerciseblock}
  \begin{problem}

  \begin{equation}
    \label{eq:normalization}
    \langle\psi_n,\psi_n\rangle=1
  \end{equation}
  When a function satisfies \eqref{eq:normalization}, we say that the function has \textit{unit length}. 
\end{problem}

\end{problem}

\subsection{Expanding the 3D equations}
Since the eigenbasis is complete, we can expand $u$, $v$, and $\Phi$ in this basis:

\begin{subequations}
  \label{eq:expanded_vars}
  \begin{align}
u(x,y,\z,t) &= \sum_{n=0}^\infty \hat{u}_n(x,y,t)\psi_n({\z}) \\
v(x,y,\z,t) &= \sum_{n=0}^\infty \hat{v}_n(x,y,t)\psi_n({\z}) \\
\Phi(x,y,\z,t) &= \sum_{n=0}^\infty \hat{\Phi}_n(x,y,t)\psi_n({\z})
\end{align}
\end{subequations}
Where 
\begin{subequations}
  \label{eq:expansion_coeffs}
\begin{align}
\hat{u}_n(x,y,t) &= \langle u, \psi_n\rangle \\
\hat{v}_n(x,y,t) &= \langle v, \psi_n\rangle \\
\hat{\Phi}_n(x,y,t) &= \langle \Phi, \psi_n\rangle
\end{align}
\end{subequations}
Using \eqref{eq:expanded_vars}, we can write \eqref{eq:linear_system_reduced_a} as 
\begin{equation}
  \frac{\partial}{\partial t}\left(\sum_{n=0}^\infty \hat{u}_n\psi_n\right) - f\left(\sum_{n=0}^\infty \hat{v}_n\psi_n\right) = -\frac{\partial}{\partial x}\sum_{n=0}^\infty\Phi_n\psi_n
\end{equation}
We can re-arrange this to give us
\begin{equation}
  \label{eq:expanded_u}
  \sum_{n=0}^\infty\left(\frac{\partial \hat{u}_n}{\partial t} - f\hat{v}_n +\frac{\partial\hat{\Phi}_n}{\partial x}\right)\psi_n=0
\end{equation}
A similar expansion of \eqref{eq:linear_system_reduced_b} gives us
\begin{equation}
    \label{eq:expanded_v}
  \sum_{n=0}^\infty\left(\frac{\partial \hat{v}_n}{\partial t} + f\hat{u}_n +\frac{\partial\hat{\Phi}_n}{\partial y}\right)\psi_n=0
\end{equation}
Expanding \eqref{eq:linear_system_reduced_c}, 
\begin{equation}
  \label{eq:expanded_vertical}
  \frac{\partial}{\partial t}L\left[\sum_{n=0}^\infty\hat\Phi_n\psi_n \right] = \frac{\partial}{\partial x}\sum_{n=0}^\infty\hat{u}_n\psi_n + \frac{\partial}{\partial y}\sum_{n=0}^\infty\hat{v}_n\psi_n
\end{equation}
Using the facts that $L[\hat\Phi_n\psi_n]=\hat\Phi_nL[\psi_n]$ and $L[\psi_n]=-\lambda_n\psi_n$, we can simplify \eqref{eq:expanded_vertical} to
\begin{equation}
  \sum_{n=0}^\infty\left(-\lambda_n\frac{\partial\hat\Phi_n}{\partial t}
  -\frac{\partial\hat u_n}{\partial x}
  - \frac{\partial\hat v_n}{\partial y} \right)\psi_n = 0
\end{equation}
Using \eqref{eq:lin_indep}, we can conclude that each parenthetical quantity in \eqref{eq:expanded_u}, \eqref{eq:expanded_v}, and \eqref{eq:expanded_vertical} is zero: for $n=0, 1, 2, ...$, we then have
%
\begin{subequations}
\label{eq:modal_system}
\begin{empheq}[box=\fbox]{align}
\frac{\partial\hat u_n}{\partial t}-f\hat v_n &= -\frac{\partial\hat\Phi_n}{\partial x} \\
\frac{\partial\hat v_n}{\partial t}+f\hat u_n &= -\frac{\partial\hat\Phi_n}{\partial y} \\
\frac{\partial\hat\Phi_n}{\partial t}+\frac{1}{\lambda_n}\left(\frac{\partial\hat u_n}{\partial x}+\frac{\partial\hat v_n}{\partial y}\right) &= 0
\label{eq:modal_system_phi}
\end{empheq}
\end{subequations}

The system \eqref{eq:modal_system} is, like \eqref{eq:swe_system}, formally similar to the shallow water equations. However, we have an infinite family of such systems, each corresponding to a given eigenfunction. Each vertical mode propagates horziontally at a phase speed $c_n=1/\sqrt{\lambda_n}=\sqrt{gh_{e,n}}$, where $h_{e,n}=1/g\lambda$ is the \textit{equivalent depth} corresponding to the eigenfunction $\psi_n$. 
\subsection{Interpreting the vertical modes}

\pianomarker

What we have done is shown that our fully 3D systems \eqref{eq:linear_system} can be understood as a linear combination of 2D shallow water systems. Each one of these shallow water systems corresponds to a particular vertical structure, given by an eigenfunction of our vertical structure operator $L$. 
\par A good analogy to what we have done is to think of a piano. A piano is capable of producing arbitrarily complicated music. However, no matter how complicated the music, we can always express the music as a combination of different piano keys being played in different combinations. In the same way, the arbitrarily complicated dynamics of our full 3D system \eqref{eq:linear_system} (the piano) can be expressed as the combinations of 2D systems (the piano keys). Furthermore, the piano keys are independent, in the sense that one cannot make the piano play, say, middle C by hitting some combination of other keys. This corresponds to the linear independence of our vertical modes \eqref{eq:lin_indep}. We will explore the extent to which this assumption, which is the property that makes our model a \textit{linear} model, is justified when considering the real atmosphere in a later point in our notes.
\marginnote{\springsmodes}
\par It's worth noting that this type of analysis is by no means unique to atmospheric science. The idea of decomposing a complicated system into a sum of independent, simpler pieces arises across physics. The canonical example is a pair of masses coupled by springs (two masses connected to each other and to two walls by three springs). Its motion can always be written as a combination of two modes: a low-frequency mode in which both masses oscillate in phase, moving together, and a higher-frequency mode in which they oscillate out of phase, moving toward and away from each other. Each mode oscillates at its own frequency, independently of the other — which is precisely the finite-dimensional version of what we have found here, where each vertical mode $\psi_n$ evolves as its own independent shallow-water system. The only difference is that the spring system has two degrees of freedom and hence two modes, whereas our operator $L$ has infinitely many. This method is refered to as \textit{normal mode analysis}, with the corresponding eigenfunctions being called \textit{normal modes}. This motivates the terminomology of calling our the $n^{th}$ eigenfunction $\psi_n$ as the $n^{th}$ \textit{vertical normal mode}.

\section{Computing the Eigenfunctions}
In this section we will actually compute the normal modes (eigenfunctions of $L$). We will first discuss the barotropic mode, which can be computed by the time-honored method of ``calculation by inspection". Next, we will consider the case where $N^2(\z)$ is a constant function. Finally, we will discuss how these normal modes can be computed numerically for a given (discrete) $N^2$ profile, as might be obtained from reanalysis or a similar gridded product. 
\subsection{The barotropic mode $\psi_0$}
We are interested in finding functions $\psi(\z)$ such that 
\begin{equation}
  \label{ex:eigenproblem}
  L[\psi(\z)] = \frac{1}{\rho_0}\frac{\partial}{\partial\z}\left(\frac{\rho_0}{N^2}\frac{\partial\psi(\z)}{\partial\z}\right) = -\lambda \psi(\z)
\end{equation}
for some number $\lambda\in\mathbb{R}$. I've made the dependence of $\psi$ on $\z$ explicit to highlight that we seek \textit{vertical functions} that can satisfy this equation. We know from Exercise \ref{ex:nonnegative_eigenvalues} that any such $\lambda\geq0$. Guided by this, we may ask: \textbf{is there any function that has an eigenvalue of $\lambda=0$?} 
\par In fact, there is! Inspection of our operator $L$ shows that any constant function $\psi(\z)=C$ would satisfy \ref{ex:eigenproblem}, with $\lambda=0$. Since we can always scale our eigenfunctions to have unit length (Exercise \ref{ex:normalization}), we take our eigenfunction corresponding to $\lambda_0$ to be 
\begin{equation}
  \label{ex:barotropic_mode}
  \boxed{
    \psi_0(\z) = \frac{1}{\sqrt{\int_{\z_B}^{\z_T}\rho_0(\z)\ d\z}} = \text{Constant for all $\z$}
  }
\end{equation}. What can we say about this mode?
\begin{itemize}
  \item \textbf{This mode is alway present:} For any basic state profiles $\rho_0(\z)$ and $N^2(\z)$, $\psi_0$ is an eigenfunction with eigenvalue $\lambda_0=0$.
  \item \textbf{This mode is the lowest mode $\psi_0$} From Exercise \ref{ex:nonnegative_eigenvalues} we know that $\lambda=0$ is the lowest value that an eigenvalue can take. Thus, $\lambda_0=0$ is the lowest eigenvalue of $L$, and $\psi_0$ can correspondingly be called the ``gravest" normal mode of $L$.
  \item \textbf{This mode has no vertical structure:} Since $\psi_0$ is a constant for all heights, our corresponding horizontal wind field is given by
  \begin{equation}
      u_0(x, y, \z, t) = \hat{u}_0(x,y,t)\psi_0(\z) = \text{Constant}\times\hat{u}_0(x,y,t)
  \end{equation}
  A similar argument for $v$ then implies 
\begin{equation}
  \frac{\partial u}{\partial \z}= \frac{\partial v}{\partial \z}=0 \iff \text{ No vertical shear in horizontal winds}
\end{equation}
Thus, our horizontal wind field that corresponds to this normal mode is constant with height. 
\item \textbf{This mode is barotropic:} The corresponding geopotential field for $\psi_0$  is given by 
\begin{equation}
  \Phi_0(x,y,\z,t)=\hat\Phi_0(x,y,t)\psi_0(\z) = \text{Constant}\times\hat{\Phi}_0(x,y,t)
\end{equation}
By the hydrostatic relation \eqref{eq:hydrostatic}, this implies that there are no temperature perturbations, so the temperature depends only on $\z$. Since $\z$ is a monotonic function of pressure, the implies that the temperature also depends only on pressure. This is the definition of a barotropic state.

\item \textbf{This mode has no vertical motion:} By the hydrostatic equation \eqref{eq:hydrostatic}, 
\begin{equation}
  \w \propto \frac{\partial}{\partial t}\left(\frac{\partial\Phi}{\partial \z}\right)
\end{equation}
Since corresponding geopotential field for $\psi_0$  is given by 
\begin{equation}
\Phi_0=\hat\Phi_0(x,y,t)\psi_0(\z),
\end{equation}
the corresponding vertical velocity $\w_0$ is given by
\begin{equation}
  \w_0\propto\frac{\partial\hat\Phi_0}{\partial t}\times\frac{\partial\psi_0}{\partial\z}=0
\end{equation}
Thus the barotropic mode corresponds to motions that are purely horizontal and without any vertical component.
\item \textbf{This mode corresponds to an infinite equivalent depth}: We defined the \textit{equivalent depth} of a vertical mode as the depth of a shallow water system that posses equal wave propagation speed \eqref{eq:equivalent_depth}. When $\lambda=0$, the corresponding equivalent depth, and thus wave speed, approaches infinity. A way to interpret this peculiar result is to say that any horizontal perturbations in the geopotential field are smoothed infinitely fast by this mode. In other words, this mode cannot support propagating gravity waves at all. 
\par An equivalent way of viewing this is to re-write \eqref{eq:modal_system_phi} as 
\begin{equation}
  \lambda_n\frac{\partial\Phi_n}{\partial t} = -\left(\frac{\partial \hat u_n}{\partial x} + \frac{\partial \hat v_n}{\partial y}\right)
\end{equation}
For $\lambda_0=0$, we thus obtain that the flow is horizontally non-divergent. A layer of fluid that is infinitely deep cannot be stretched or squeezed by any finite amount, and is thus non-divergent; this gives us the $h_e\to\infty$ result.
\end{itemize}

\subsection{The baroclinic modes for constant $N^2$}
For modes that are higher order than the barotropic mode ($n>0$), we have
\begin{equation}
  \frac{\partial\Phi_n}{\partial \z}=\hat\Phi_n(x,y,t)\frac{\partial\psi_n}{\partial\z}\neq 0 
\end{equation}
Thus, by the hydrostatic equation \eqref{eq:hydrostatic}, we have non-zero temperature perturbation from the basic state temperature profile $T_0(\z)$ along a given surface $\z=\text{ const.}$ Therefore, temperature depends both on pressure (through $\z$) and on density (by the ideal gas law). This is the definition of a baroclinic state, and motivates us to call the vertical normal mode $\psi_{n\geq1}$ as the $n^{\text{th}}$ \textit{baroclinic mode}. 
\par In general, since $L$ (and thus its eigenfunctions) depend on $N^2(\z)$, we cannot compute a closed form expression for the baroclinic modes. However, in the case that $N^2(\z)$ is a constant, then we can obtain analytical expressions for the baroclinic modes.
\par For a constant $N^2$, our eigenproblem \eqref{ex:eigenproblem} becomes
\begin{equation}
  \label{eq:constant_N2_eigenproblem}
  L[\psi] = \frac{1}{\rho_0N^2}\frac{\partial}{\partial \z}\left(\rho_0\frac{\partial\psi}{\partial\z}\right)=-\lambda\psi \qquad \text{($N^2 = $ const)}
\end{equation}
%
By the product rule, \eqref{eq:constant_N2_eigenproblem} becomes
\begin{equation}
  \frac{1}{\rho_0N^2}\left(\frac{\partial\rho_0}{\partial\z}\frac{\partial\psi}{\partial\z}+\rho_0\frac{\partial^2\psi}{\partial\z}\right)
\end{equation}





\section{Numerical Implementation}
We now consider the question of how to apply the ideas explored in the previous section to actual data. To do this, we will convert the operators (specifically $L[\Phi]$ and $\langle\cdot,\cdot\rangle$) and functions of the previous section into matrices and vectors, respectively. 

\subsection{Discretizing the vertical structure operator and the stiffness matrix}

Let's begin again with the vertical structure operator
\begin{equation}
  L[\Phi] = \frac{1}{\rho_0}\frac{\partial}{\partial\z}\left(\frac{\rho_0}{N^2}\frac{\partial\Phi}{\partial\z}\right)
\end{equation}
with boundary conditions
\begin{equation}
 \left. \frac{\partial\Phi}{\partial\z}\right|_{\z_B}=\left. \frac{\partial\Phi}{\partial\z}\right|_{\z_T} = 0
\end{equation}

To make the following discussion a little easier to read, and since we anticipate working with observational data that is typically reported in pressure coordinates, we can re-write the same operator in isobaric coordinates as 
\begin{equation}
  L[\phi] = \frac{\partial}{\partial p}\left(\frac{p}{SR_d}\frac{\partial\phi}{\partial p}\right)
\end{equation}
with boundary condition
\begin{equation}
  \label{eq:L_in_p}
 \left. \frac{\partial\phi}{\partial p}\right|_{p_B}=\left. \frac{\partial\phi}{\partial p}\right|_{p_T} = 0
\end{equation}
Where $\phi(p)$ is the geopotential in pressure coordinates, and $S=-\frac{T}{\theta}\frac{\partial \theta}{\partial p}$ is the \textit{static stability}. 

\begin{exercise}
  Show that $L[\Phi]$ as given in \eqref{eq:L_def} in log-p height coordinates $\z$ is equivalent to Equation \ref{eq:L_in_p} when converted to pressure coordinates. 
  \\
  \textbf{Hint:} Recall that $\partial p / \partial \z = - p/H_s$, where $H_s$ is a constant scale height
\end{exercise}

\begin{exercise}
  Show that $L[\phi]$ is self-adjoint with respect to the following inner product:
  $$
  \langle f, g\rangle = \int_{p_T}^{p_B}f(p)g(p)dp
  $$
\end{exercise}

Defining $a(p)=p/SR_d$ for convenience, our eigenvalue problem 
$$
L[\phi]=-\lambda \phi
$$

becomes
\begin{equation}
\label{eq:eigenproblem_in_in}
\partial_p(a\partial_p\phi) = -\lambda\phi
\end{equation}

To discretize this problem, it is not enough to simply replace the derivatives with finite differences; the reason is that our operator is self-adjoint with respect to a \textit{particular} inner product. The self-adjointness lives in the pairing of the operator with the inner product, not in the operator alone. To preserve that on the discrete side, you have to discretize them together.

\par To pair the operator and inner product together, we multiply both sides of Equation \ref{eq:eigenproblem_in_in} by some ``test function'' $v(p)$ and integrate over $[p_T, p_B]$:

$$
\int_{p_T}^{p_B}v\partial_p(a\partial_p\phi) = -\lambda\int_{p_T}^{p_B}v\phi dp
$$
Integrating by parts on the LHS and utilizing boundary conditions gives us

\begin{equation}
  \label{eq:weak_form}
  \int_{p_T}^{p_B}a(p)\partial_pv\partial_p\phi dp = \lambda\int_{p_T}^{p_B}v\phi dp
\end{equation}

Where primes indicate derivatives. This is the so-called \textit{weak form} of the eigenvalue problem, so called because we require two integrals to be equal, rather than two functions. The RHS involves the inner product of $\phi$ with our test function $v$, and the LHS is a functional $k(\phi, v)$ that takes in two functions $\phi$ and $v$:
$$
k(\phi, v) = \int_{p_T}^{p_B}a(p)\partial_pv\partial_p\phi dp
$$
The objective is now to discretize $k$ and $\langle\cdot,\cdot\rangle$ as matrices acting on finite-dimensional vectors. To do this, we choose a finite set of $n$ basis functions $\eta_j$ to approximate $\phi$:
$$
\phi(p) \approx \sum_{j=0}^{n-1}\phi_j\eta_j(p)
$$

We then require that Equation \ref{eq:weak_form} hold for any test function that is also one of our basis functions $\eta_i$:
\begin{equation}
  \label{eq:matrix_elements}
  \sum_{j=0}^{n-1} \phi_j \underbrace{\int a(p)\, \eta_i'(p)\, \eta_j'(p)\, dp}_{K_{ij}} = \lambda   \sum_{j=0}^{n-1}\phi_j \underbrace{\int \eta_i(p)\, \eta_j(p)\, dp}_{M_{ij}}
\end{equation}

This gives us $n$ equations for each choice of $i=0,...,n-1$. We can express this system as the matrix equation 
\begin{equation}
  \label{eq:matrix_equation}
  K\phi = \lambda M\phi
\end{equation}
Where each entry at row, column ($i$,$j$) of the matricies $M$ and $K$ are given by the indicated integral expressions in Equation \ref{eq:matrix_elements}. Equation \ref{eq:matrix_equation} is known as the \textit{generalized eigenvalue problem}, and holds for any choice of basis function functions $\{\eta_i\}$. In the next section we will describe a particularly convenient family of basis functions, which will make use of a discretized pressure coordinate grid.

To specify our basis functions concretely, we first discretize the
pressure interval. Choose $n$ pressure levels
\[
  p_T = p_0 < p_1 < \cdots < p_{n-1} = p_B,
\]
not necessarily evenly spaced. In practice, these will be the levels at
which our data is reported. We will refer to the spacing between adjacent
levels as $\Delta p_j := p_{j+1} - p_j$.

Associated with this grid, we define the \emph{hat functions} $\eta_i(p)$:
each $\eta_i$ is the unique continuous, piecewise-linear function on
$[p_T, p_B]$ satisfying
\[
  \eta_i(p_j) =
  \begin{cases}
    1, & i = j,\\
    0, & i \neq j.
  \end{cases}
\]
By construction, $\eta_i$ has value $1$ at the grid point $p_i$, value $0$ at every other grid point, and is linearly interpolated between them.  From this definition, the derivative of each $\eta$ is piecewise constant:

\[
  \eta_i'(p) =
  \begin{cases}
    +1/\Delta p_{i-1}, & p\in[p_{i-1}, p_i] ,\\
    -1/\Delta p_{i}, & p\in[p_{i}, p_{i+1}] ,\\
    0, & \text{otherwise}
  \end{cases}
\]

We can now directly compute the elements of $M$ and $K$. 

For the diagonal elements of $K$, 
\begin{equation}
  K_{ii}=\int_{p_{i-1}}^{p_i}\frac{a}{\Delta p_{i-1}^2}dp + \int_{p_{i}}^{p_{i+1}}\frac{a}{\Delta p_{i}^2}dp \approx \frac{a_{i-1/2}}{\Delta p_{i-1}} + \frac{a_{i+1/2}}{\Delta p_i}
\end{equation}
Where we have made approximated $a(p)$ over each interval $[p_k, p_{k+1}]$ by its midpoint value, denoted
$$
a_{k+1/2}=\frac{a_k+a_{k+1}}{2}
$$

For off diagonal elements, we note that only entries one-off the diagonal ($j=i\pm1$) are non-zero since hat functions only overlap with neighbors:
\begin{equation}
  K_{i, i-1}=\int_{p_{i-1}}^{p_i}a(p)\times\frac{1}{\Delta p_{i-1}}\times\left(\frac{-1}{\Delta p_{i-1}}\right)dp \approx -\frac{a_{i-1/2}}{\Delta p_{i-1}}
\end{equation}
And a similar calculation yields 
\begin{equation}
K_{i, i+1} \approx -\frac{a_{i+1/2}}{\Delta p_{i}}
\end{equation}

Computing the elements of $M$ involves the hat functions themselves, and not their derivatives (Equation \ref{eq:matrix_elements}). For diagonal elements, 
\begin{equation}
  M_{ii}=\int_{p_{i}}^{p_{i+1}}\eta_i^2dp = \frac{\Delta p_{i-1}+\Delta p_i}{3}
\end{equation}
And the one-off diagonal entries (for the same reason as before) are given by
\begin{equation}
  M_{i, i+1}=M_{i+1, i}=\int_{p_{i}}^{p_{i+1}}\eta_i\eta_{i+1}dp = \frac{\Delta p_i}{6}
\end{equation}
